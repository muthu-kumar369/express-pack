# Task 5: OpenAPI/Swagger Integration

## Context

express-pack currently has no built-in support for API documentation via OpenAPI/Swagger. Developers must manually write and maintain OpenAPI specifications, which often become outdated as the API evolves.

## Objective

Integrate OpenAPI/Swagger generation directly into express-pack, allowing automatic API documentation generation from routes, validation schemas, and TypeScript types.

**1. generator.ts:**
```typescript
import swaggerJsdoc from 'swagger-jsdoc';
import { Application } from 'express';
import { RouteGroup } from '../../framework/express/types';

export interface OpenAPIConfig {
  enabled?: boolean;
  output?: string;
  ui?: string | false;
  info?: {
    title?: string;
    version?: string;
    description?: string;
  };
  servers?: Array<{
    url: string;
    description?: string;
  }>;
  security?: any[];
}

/**
 * Generates OpenAPI specification from routes
 */
export class OpenAPIGenerator {
  private config: OpenAPIConfig;
  private spec: any;

  constructor(config: OpenAPIConfig = {}) {
    this.config = {
      enabled: true,
      output: './openapi.json',
      ui: '/api-docs',
      info: {
        title: 'API Documentation',
        version: '1.0.0',
        description: 'API documentation generated by express-pack',
      },
      servers: [
        { url: 'http://localhost:3000', description: 'Development server' },
      ],
      ...config,
    };
  }

  /**
   * Generate OpenAPI spec from routes
   */
  generate(routes: RouteGroup[]): any {
    const paths: any = {};
    const components: any = {
      schemas: {},
      securitySchemes: {
        bearerAuth: {
          type: 'http',
          scheme: 'bearer',
          bearerFormat: 'JWT',
        },
      },
    };

    // Extract paths from routes
    routes.forEach(({ prefix = '', version = '', route: routeList = [] }) => {
      const basePath = prefix + version;

      routeList.forEach(({ path, route }) => {
        const fullPath = basePath + path;
        paths[fullPath] = this.extractRouteInfo(route);
      });
    });

    this.spec = {
      openapi: '3.0.0',
      info: this.config.info,
      servers: this.config.servers,
      paths,
      components,
    };

    return this.spec;
  }

  /**
   * Extract route information from Express router
   */
  private extractRouteInfo(router: any): any {
    const methods: any = {};

    // Iterate through router stack
    router.stack?.forEach((layer: any) => {
      if (layer.route) {
        const { path, methods: routeMethods } = layer.route;

        Object.keys(routeMethods).forEach((method) => {
          if (routeMethods[method]) {
            methods[method] = this.generateMethodDoc(layer, method);
          }
        });
      }
    });

    return methods;
  }

  /**
   * Generate documentation for a specific method
   */
  private generateMethodDoc(layer: any, method: string): any {
    // Extract from metadata if available
    const metadata = layer.route?.metadata || {};

    return {
      summary: metadata.summary || `${method.toUpperCase()} endpoint`,
      description: metadata.description || '',
      tags: metadata.tags || [],
      parameters: this.extractParameters(metadata),
      requestBody: this.extractRequestBody(metadata),
      responses: this.extractResponses(metadata),
      security: metadata.requiresAuth ? [{ bearerAuth: [] }] : undefined,
    };
  }

  /**
   * Extract parameters from metadata
   */
  private extractParameters(metadata: any): any[] {
    const params: any[] = [];

    // Path parameters
    if (metadata.params) {
      Object.keys(metadata.params).forEach((name) => {
        params.push({
          name,
          in: 'path',
          required: true,
          schema: this.zodToOpenAPI(metadata.params[name]),
        });
      });
    }

    // Query parameters
    if (metadata.query) {
      Object.keys(metadata.query).forEach((name) => {
        params.push({
          name,
          in: 'query',
          required: false,
          schema: this.zodToOpenAPI(metadata.query[name]),
        });
      });
    }

    return params;
  }

  /**
   * Extract request body from metadata
   */
  private extractRequestBody(metadata: any): any {
    if (!metadata.body) return undefined;

    return {
      required: true,
      content: {
        'application/json': {
          schema: this.zodToOpenAPI(metadata.body),
        },
      },
    };
  }

  /**
   * Extract responses from metadata
   */
  private extractResponses(metadata: any): any {
    return {
      200: {
        description: 'Successful response',
        content: {
          'application/json': {
            schema: metadata.response || { type: 'object' },
          },
        },
      },
      400: {
        description: 'Bad request',
        content: {
          'application/json': {
            schema: {
              type: 'object',
              properties: {
                success: { type: 'boolean', example: false },
                error: { type: 'string' },
                code: { type: 'string' },
              },
            },
          },
        },
      },
      401: {
        description: 'Unauthorized',
      },
      500: {
        description: 'Internal server error',
      },
    };
  }

  /**
   * Convert Zod schema to OpenAPI schema
   */
  private zodToOpenAPI(zodSchema: any): any {
    // Simplified conversion - in real implementation, use zod-to-openapi
    if (!zodSchema) return { type: 'object' };

    // This is a placeholder - actual implementation would parse Zod schema
    return {
      type: 'object',
      properties: {},
    };
  }

  /**
   * Save spec to file
   */
  saveToFile(filepath?: string): void {
    const fs = require('fs');
    const path = filepath || this.config.output;
    fs.writeFileSync(path, JSON.stringify(this.spec, null, 2));
  }
}
```

```typescript
export function setupSwaggerUI(
  app: Application,
  spec: any,
  path: string = '/api-docs'
): void {
  app.use(path, swaggerUi.serve, swaggerUi.setup(spec, {
    customCss: '.swagger-ui .topbar { display: none }',
    customSiteTitle: 'API Documentation',
```

```typescript
/**
 * Decorator to add OpenAPI metadata to routes
 */
export function ApiDoc(metadata: {
  summary?: string;
  description?: string;
  tags?: string[];
  params?: any;
  query?: any;
  body?: any;
  response?: any;
  requiresAuth?: boolean;
}) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    // Attach metadata to route handler
    if (!descriptor.value.metadata) {
      descriptor.value.metadata = {};
    }
    Object.assign(descriptor.value.metadata, metadata);
    return descriptor;
  };
}
```

### 3. Integrate with ExpressPack

**Update framework/express/index.ts:**
```typescript
import { OpenAPIGenerator, setupSwaggerUI, OpenAPIConfig } from '../../util/openapi';

export class ExpressPack {
  // ... existing code

  static initRoutes({ 
    routes = [], 
    openapi 
  }: { 
    routes?: RouteGroup[]; 
    openapi?: OpenAPIConfig;
  }) {
    if (!this.#initialized || !this.#app) {
      throw new Error('Cannot bind routes before app initialization.');
    }

    // Bind routes
    routes.forEach(({ prefix = '', version = '', route: routeList = [] }) => {
      const basePath = prefix + version;

      routeList.forEach(({ path, route }) => {
        const fullPath = basePath ? basePath + path : path;
        this.#app!.use(fullPath, route);
      });
    });

    // Generate OpenAPI if enabled
    if (openapi?.enabled) {
      const generator = new OpenAPIGenerator(openapi);
      const spec = generator.generate(routes);

      // Save to file if output specified
      if (openapi.output) {
        generator.saveToFile(openapi.output);
      }

      // Setup Swagger UI if path specified
      if (openapi.ui) {
        setupSwaggerUI(this.#app, spec, openapi.ui);
      }
    }
  }
}
```
